cmake_minimum_required(VERSION 3.15)
project(SupersonicEngine C CXX)

# Require C++17 standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Compile as a Shared Library (DLL) named opengl32
add_library(opengl32 SHARED
    src/dllmain.cpp
    src/d3d12_hook.cpp
)

# Link Windows libraries needed for D3D12 and DXGI
target_link_libraries(opengl32
    d3d12.lib
    dxgi.lib
    user32.lib
)
