cmake_minimum_required(VERSION 3.15)
project(SupersonicEngine CXX)

# Set C++ Standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED True)

# Create a shared library named 'opengl32'
add_library(opengl32 SHARED src/dllmain.cpp)

# Link Windows libraries required for Direct3D 12 and OpenGL Proxy
target_link_libraries(opengl32 PRIVATE d3d12 dxgi user32 gdi32)
